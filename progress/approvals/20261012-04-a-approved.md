### 核心架构

```

│                       Agent Runner                          │
├─────────────────────────────────────────────────────────────┤
│  1. 认领租约 (fencing_token++)                              │

└─────────────────────────────────────────────────────────────┘
```

### Fencing 机制

```python
class FencedPostgresSaver(PostgresSaver):
    async def put(self, config, checkpoint, metadata):
        async with self.db.transaction():
            # 1. 锁定 Run + Lease
            run = await self.db.select_for_update(
                "run_execution", run_id=config["run_id"]
            )
            lease = await self.db.select_for_update(
                "execution_lease", run_id=config["run_id"]
            )
            
            # 2. Fencing 检查
            if lease.fencing_token != config["expected_token"]:
                raise FencingError("Stale executor")
            if lease.expires_at < now():
                raise FencingError("Lease expired")
            if run.state_version != config["expected_version"]:
                raise FencingError("Version mismatch")
            
            # 3. 写入 checkpoint (加密)
            encrypted = await self.encrypt(checkpoint, run.key_ref)
            checkpoint_id = await self.framework_store.save(encrypted)
            
            # 4. 更新绑定
            await self.db.upsert("checkpoint_binding", {
                "run_id": run.run_id,
                "checkpoint_id": checkpoint_id,
                "state_version": run.state_version,
                "fencing_token": lease.fencing_token,
            })
```

### 结果提交 UoW

```python
async def commit_result(
    token: LeaseToken,
    expected_version: int,
    result: VerifiedCandidate
) -> ResultReceipt:
    async with db.transaction():
        # 1. Fencing
        await check_lease_and_version(token, expected_version)
        
        # 2. 重新鉴权
        await recheck_authorization(run, result.evidence_ids)
        
        # 3. 检查幂等
        if existing := await get_result_commit(run.run_id):
            if same_result(existing, result):
                return existing.receipt
            else:
                raise ConflictError("Different result")
        
        # 4. 原子写入（全部成功或全部失败）
        await insert_claims(result.claims)          # Claims 表
        await insert_citations(result.citations)    # Citations 表
        await update_message(result.content)        # Messages 表
        await update_run_state(result.terminal)     # Runs 表
        await insert_result_commit(result_id)       # result_commit 表
        await insert_outbox_events([                # outbox 表
            ("token", seq=1, ...),
            ("citation", seq=2, ...),
            ("completed", seq=3, ...)
        ])
        
        # 5. 递增 state_version
        await increment_state_version(run.run_id)
        
        return ResultReceipt(...)
```

### 初始化脚本（将提供）

```bash
# 启用 KV secrets engine
openbao secrets enable -path=secret kv-v2

# 创建 checkpoint 密钥
openbao kv put secret/pivot/checkpoint-key \
  key=$(openssl rand -base64 32) \
  version=1

# 创建 admission 密钥
openbao kv put secret/pivot/admission-key \
  key=$(openssl rand -base64 32) \
  version=1

# 创建应用策略
openbao policy write pivot-runner - <<EOF
path "secret/data/pivot/*" {
  capabilities = ["read"]
}
EOF

# 创建 token
openbao token create -policy=pivot-runner
```
